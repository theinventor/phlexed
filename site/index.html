With phlexed app/views/settings/index_view.rb
24 lines · composed components · registry-aware 
module Settings
  class IndexView < Views::Base
    def initialize(user:)
      @user = user
    end

    def view_template
      h1(class: "text-3xl font-bold mb-6") { "Settings" }

      render PhlexyUI::Card.new(bordered: true, class: "max-w-2xl") do
        render PhlexyUI::CardTitle.new { "Profile" }
        render PhlexyUI::CardBody.new do
          para(class: "text-base-content/70") { "Update your public profile." }

          form_with(model: @user, url: settings_path, method: :patch) do |f|
            render PhlexyUI::FormControl.new(label: "Name")  { f.text_field :name,  class: "input input-bordered w-full" }
            render PhlexyUI::FormControl.new(label: "Email") { f.email_field :email, class: "input input-bordered w-full" }
            render PhlexyUI::FormControl.new(label: "Bio")   { f.text_area :bio,    class: "textarea textarea-bordered w-full", rows: 4 }
            render PhlexyUI::Toggle.new(label: "Receive notifications", variant: :primary, name: "user[notifications_enabled]")

            render PhlexyUI::CardActions.new(justify: :end) do
              render PhlexyUI::Button.new(as: :a, variant: :ghost, href: root_path) { "Cancel" }
              render PhlexyUI::Button.new(as: :submit, variant: :primary) { "Save" }
            end
          end
        end
      end
    end
  end
end
  • ✓ Composed from registered PhlexyUI components
  • ✓ Theme-switch safe — no hardcoded colors
  • ✓ Every piece is reusable on the next page

module Pricing
  class IndexView < Views::Base
    PLANS = [
      { name: "Starter", price: "$0",  features: ["1 project", "Community support"],                          cta: "Get started", variant: :outline },
      { name: "Pro",     price: "$29", features: ["Unlimited projects", "Priority support", "All integrations"], cta: "Start trial", variant: :primary, featured: true },
      { name: "Team",    price: "$99", features: ["Everything in Pro", "SSO + SAML", "Dedicated CSM"],            cta: "Contact sales", variant: :outline }
    ]

    def view_template
      section(class: "py-16 bg-base-200") do
        h1(class: "text-4xl font-bold text-center mb-12") { "Pricing" }

        div(class: "grid md:grid-cols-3 gap-6 max-w-5xl mx-auto px-4") do
          PLANS.each do |plan|
            render PhlexyUI::Card.new(bordered: true, variant: plan[:featured] ? :primary : nil) do
              render PhlexyUI::CardBody.new do
                render PhlexyUI::Badge.new(variant: :accent) { "Most popular" } if plan[:featured]
                render PhlexyUI::CardTitle.new { plan[:name] }
                para(class: "text-3xl font-bold") { "#{plan[:price]}" + plain(" /mo") }
                ul(class: "mt-4 space-y-1 text-sm") { plan[:features].each { |f| li { "• #{f}" } } }
                render PhlexyUI::CardActions.new do
                  render PhlexyUI::Button.new(variant: plan[:variant], class: "w-full") { plan[:cta] }
                end
              end
            end
          end
        end
      end
    end
  end
end
✓ Plans are data, not markup
✓ Add a 4th tier in one line
✓ Theme-switch safe

app/views/dashboard/show_view.rb 
module Dashboard
  class ShowView < Views::Base
    def initialize(stats:, activities:)
      @stats = stats
      @activities = activities
    end

    def view_template
      div(class: "p-8 bg-base-200 min-h-screen") do
        h1(class: "text-3xl font-bold mb-6") { "Dashboard" }

        render PhlexyUI::Stats.new(shadow: true, class: "w-full mb-8") do
          @stats.each do |s|
            render PhlexyUI::Stat.new(
              figure: s[:icon],
              figure_color: s[:color],
              title: s[:title],
              value: s[:value],
              value_color: s[:color],
              desc: s[:desc]
            )
          end
        end

        render PhlexyUI::Card.new(bordered: true) do
          render PhlexyUI::CardBody.new do
            render PhlexyUI::CardTitle.new { "Recent activity" }
            ul(class: "divide-y divide-base-300") do
              @activities.each do |a|
                li(class: "py-3 flex justify-between") do
                  span { a.description }
                  span(class: "text-sm text-base-content/60") { time_ago_in_words(a.created_at) }
                end
              end
            end
          end
        end
      end
    end
  end
end

app/views/shared/nav.rb 
module Shared
  class Nav < Views::Base
    def initialize(user:, links:)
      @user = user
      @links = links
    end

    def view_template
      render PhlexyUI::Navbar.new(bordered: true) do |nav|
        nav.start { render PhlexyUI::Button.new(as: :a, variant: :ghost, href: "/", class: "text-xl") { "Acme" } }

        nav.center do
          render PhlexyUI::Menu.new(horizontal: true) do
            @links.each { |l| li { a(href: l[:href]) { l[:label] } } }
          end
        end

        nav.end_ do
          render Shared::ThemeToggle.new
          render PhlexyUI::Dropdown.new(align: :end) do |d|
            d.trigger { render PhlexyUI::Avatar.new(src: @user.avatar_url, size: :sm, circle: true) }
            d.content do
              render PhlexyUI::Menu.new(size: :sm) do
                li { a(href: profile_path) { "Profile" } }
                li { a(href: settings_path) { "Settings" } }
                li { button_to "Sign out", sign_out_path, method: :delete }
              end
            end
          end
        end
      end
    end
  end
end

Why it works

The AI wasn’t broken. It was blind.

Phlex component libraries already exist. The AI just didn’t know about them. phlexed fixes that.

Component registry

Scans your Phlex library and builds a structured index of every available component, prop, variant, and example. Claude prompts get that context automatically.

Style registry

Reads your DaisyUI themes, Tailwind config, and custom colors. Rules go into CLAUDE.md so the AI stops writing bg-blue-500 when bg-primary exists.

Routing rules

Injects routing into CLAUDE.md so “build me a page” automatically invokes /phlexed-build with full context. No copy-pasting prompts.

Five skills, one install

A skill for every Phlex moment.

/phlexed-setup one-time

Detect, index, configure

Detects your Phlex component library, scans it into a registry, reads your DaisyUI/Tailwind config, and injects routing rules into CLAUDE.md.

/phlexed-build every day

Generate a page or feature

Plans the layout, picks registered components, generates composed Phlex views. Auto-invokes /phlexed-component if something’s missing.

/phlexed-component as needed

Create a new component

Reads your library’s conventions, generates a new Phlex component following the right patterns, adds tests, rebuilds the registry.

/phlexed-retrofit migration

Convert ERB → Phlex autonomously

Audits app/views/, batches by dependency, generates a Ralph loop, then converts one view per iteration with atomic commits and preserved behavior.

/phlexed-theme design system

Restyle without wrecking the design system

Switch DaisyUI themes, define custom themes, or restyle components — all through props and theme config. Never inline styles. Never hardcoded colors. Audit mode flags existing violations.

How it works

Four steps from install to shipping.

  1. 01

    Detect

    phlexed reads Gemfile.lock and finds your installed Phlex component library.

  2. 02

    Index

    Runs the right adapter, parses each component’s props and variants, writes .phlexed/registry.json.

  3. 03

    Configure

    Builds the style registry from your Tailwind config and appends routing + styling rules to CLAUDE.md.

  4. 04

    Prompt

    “Build me a settings page” → Claude invokes /phlexed-build with full context and ships composed Phlex.
